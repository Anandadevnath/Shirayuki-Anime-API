// Provider registry: connects dynamic config + scraper engines + adapters +
// health/managers. The single place that knows about concrete providers.
//
// A provider is ADDED here (plus one folder under src/providers/<id>/). The
// manager/routes never reference a provider by name — they only see the
// normalized interface, so a single site's breakage never reaches the frontend.
 
const engineCache = new Map(); // engineId -> engine instance
function getEngine(id) {
  if (!engineCache.has(id)) {
    if (id === 'http') engineCache.set(id, HttpScraperEngine());
    else throw new Error(`Unknown scraper engine: ${id}`);
  }
  return engineCache.get(id);
}
